import os
from datetime import datetime

from netmiko import ConnectHandler
from netmiko.exceptions import NetmikoTimeoutException, NetmikoAuthenticationException


HOSTS_FILE = "hosts.txt"
COMMANDS_FILE = "comandos.txt"
OUTPUT_FILE = "resultados.txt"

# device_type do Netmiko para HPE Comware.
DEVICE_TYPE = "hp_comware"


def ler_linhas(caminho):
    """Lê linhas não vazias e ignora comentários iniciados por #."""
    with open(caminho, "r", encoding="utf-8") as arquivo:
        return [
            linha.strip()
            for linha in arquivo
            if linha.strip() and not linha.lstrip().startswith("#")
        ]


def main():
    try:
        hosts = ler_linhas(HOSTS_FILE)
        comandos = ler_linhas(COMMANDS_FILE)
    except OSError as erro:
        print(f"Não foi possível ler um arquivo de entrada: {erro}")
        return

    if not hosts:
        print(f"Nenhum host encontrado em {HOSTS_FILE}.")
        return

    if not comandos:
        print(f"Nenhum comando encontrado em {COMMANDS_FILE}.")
        return

    # Solicita as credenciais sem gravá-las nos arquivos ou no script.
    username = input("Usuário: ")
    password = os.getenv("COMWARE_PASSWORD")

    if not password:
        import getpass
        password = getpass.getpass("Senha: ")

    data_hora = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    with open(OUTPUT_FILE, "w", encoding="utf-8") as saida:
        saida.write(f"Coleta iniciada: {data_hora}\n")

        for host in hosts:
            print(f"Conectando a {host}...")
            saida.write(f"\n{'=' * 70}\nHOST: {host}\n{'=' * 70}\n")

            dispositivo = {
                "device_type": DEVICE_TYPE,
                "host": host,
                "username": username,
                "password": password,
                "conn_timeout": 10,
                "banner_timeout": 15,
                "auth_timeout": 15,
            }

            conexao = None

            try:
                conexao = ConnectHandler(**dispositivo)

                for comando in comandos:
                    print(f"  Executando: {comando}")
                    saida.write(f"\n### COMANDO: {comando}\n")

                    try:
                        resultado = conexao.send_command(
                            comando,
                            read_timeout=30,
                        )
                        saida.write(resultado.rstrip() + "\n")
                    except Exception as erro:
                        mensagem = f"Falha ao executar {comando!r}: {erro}"
                        print(f"  {mensagem}")
                        saida.write(mensagem + "\n")

            except NetmikoAuthenticationException:
                mensagem = f"Falha de autenticação em {host}."
                print(mensagem)
                saida.write(mensagem + "\n")

            except NetmikoTimeoutException:
                mensagem = f"Tempo esgotado ao conectar a {host}."
                print(mensagem)
                saida.write(mensagem + "\n")

            except Exception as erro:
                mensagem = f"Erro ao conectar a {host}: {erro}"
                print(mensagem)
                saida.write(mensagem + "\n")

            finally:
                if conexao is not None:
                    conexao.disconnect()

    print(f"\nColeta concluída. Resultados salvos em {OUTPUT_FILE}.")


if __name__ == "__main__":
    main()
